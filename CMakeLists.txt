cmake_minimum_required(VERSION 3.16)
project(lab4_linked_lists LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(lists src/lists.cpp)
target_include_directories(lists PUBLIC include)
add_executable(lab4 src/main.cpp)
target_link_libraries(lab4 PRIVATE lists)
include(CTest)
if(BUILD_TESTING)
    find_package(GTest QUIET)
    if(GTest_FOUND)
        add_executable(list_tests tests/test_list.cpp)
        target_link_libraries(list_tests PRIVATE lists GTest::gtest_main)
        include(GoogleTest)
        gtest_discover_tests(list_tests)
    else()
        message(WARNING "Google Test not found; install libgtest-dev to enable tests")
    endif()
endif()
